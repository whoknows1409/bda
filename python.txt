================================================================================
                   PROGRAM 1: WORD COUNT
================================================================================

DATASET (wordcount_input.txt):
-------------------------------
Hello World Hello Hadoop
World is beautiful
Hadoop is powerful
Hello World

-------------------------------
MAPPER: mapper_wc.py
-------------------------------
#!/usr/bin/env python3
import sys

for line in sys.stdin:
    line = line.strip()
    words = line.split()
    for word in words:
        print(f"{word}\t1")

-------------------------------
REDUCER: reducer_wc.py
-------------------------------
#!/usr/bin/env python3
import sys

current_word = None
current_count = 0

for line in sys.stdin:
    line = line.strip()
    word, count = line.split('\t', 1)
    count = int(count)

    if current_word == word:
        current_count += count
    else:
        if current_word is not None:
            print(f"{current_word}\t{current_count}")
        current_word = word
        current_count = count

if current_word is not None:
    print(f"{current_word}\t{current_count}")

-------------------------------
STEPS TO RUN:
-------------------------------
# 1. Create input file
nano wordcount_input.txt
# (type the dataset content, save with Ctrl+O, exit with Ctrl+X)

# 2. Give execute permissions
chmod +x mapper_wc.py reducer_wc.py

# 3. Copy input to HDFS
hdfs dfs -mkdir /input/wordcount
hdfs dfs -put wordcount_input.txt /input/wordcount/

# 4. Run MapReduce job
hadoop jar /usr/local/hadoop/share/hadoop/tools/lib/hadoop-streaming-*.jar \
  -input /input/wordcount/wordcount_input.txt \
  -output /output/wordcount \
  -mapper mapper_wc.py \
  -reducer reducer_wc.py \
  -file mapper_wc.py \
  -file reducer_wc.py

# 5. View output
hdfs dfs -cat /output/wordcount/part-00000

# 6. Remove output before re-running (if needed)
hdfs dfs -rm -r /output/wordcount

-------------------------------
EXPECTED OUTPUT:
-------------------------------
Hadoop    2
Hello     3
World     3
beautiful 1
is        2
powerful  1

#!/usr/bin/env python3
import sys

current_year = None
max_temp = float('-inf')
min_temp = float('inf')

for line in sys.stdin:
    line = line.strip()
    if not line:
        continue
    parts = line.split('\t')
    if len(parts) != 2:
        continue
    year = parts[0]
    temp = float(parts[1])

    if current_year == year:
        if temp > max_temp:
            max_temp = temp
        if temp < min_temp:
            min_temp = temp
    else:
        if current_year is not None:
            print(f"{current_year}\tMaxTemp:{max_temp}\tMinTemp:{min_temp}")
        current_year = year
        max_temp = temp
        min_temp = temp

if current_year is not None:
    print(f"{current_year}\tMaxTemp:{max_temp}\tMinTemp:{min_temp}")

#!/usr/bin/env python3
import sys

current_word = None
doc_list = []

for line in sys.stdin:
    line = line.strip()
    if not line:
        continue
    parts = line.split('\t')
    if len(parts) != 2:
        continue
    word = parts[0]
    doc = parts[1]

    if current_word == word:
        if doc not in doc_list:
            doc_list.append(doc)
    else:
        if current_word is not None:
            print(f"{current_word}\t{', '.join(doc_list)}")
        current_word = word
        doc_list = [doc]

if current_word is not None:
    print(f"{current_word}\t{', '.join(doc_list)}")

-------------------------------

================================================================================
                         COMMON ERRORS & FIXES
================================================================================

ERROR: Output directory already exists
FIX:   hdfs dfs -rm -r /output/dirname

ERROR: Permission denied on mapper/reducer
FIX:   chmod +x mapper.py reducer.py

ERROR: Cannot find hadoop-streaming jar
FIX:   find /usr/local/hadoop -name "hadoop-streaming*.jar"
       Then use that full path in the hadoop jar command

ERROR: Python not found
FIX:   which python3   (use the full path in -mapper flag)
       -mapper /usr/bin/python3 mapper.py

ERROR: jps shows nothing / Hadoop not running
FIX:   stop-all.sh
       start-dfs.sh
       start-yarn.sh
       jps

ERROR: Tab vs Space in dataset
FIX:   Use \t explicitly. In nano: press Ctrl+V then Tab to insert real tab
